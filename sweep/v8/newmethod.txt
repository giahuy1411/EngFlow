    public List<Exercise> generateAll(Lesson lesson, int count) {
        int perType = Math.max(1, count / 5);
        java.util.List<java.util.List<Exercise>> buckets = java.util.List.of(
                generateMultipleChoice(lesson, perType),
                generateFillBlank(lesson, perType),
                generateMatching(lesson, perType),
                generateTranslation(lesson, perType),
                // Never negative: the old "perType + remainder" arithmetic asked the
                // listening generator for -1 items when count was below 5.
                generateListening(lesson, Math.max(1, count - 4 * perType)));
        // audit-v8: `count` is a ceiling, not a suggestion. Each generator can return
        // more than perType, so five types overshoot (measured: count=3 saved 30 rows).
        // Take one item per type in rotation until the budget is spent, so a small
        // count still covers every type instead of keeping whichever ran first.
        List<Exercise> all = new ArrayList<>();
        for (int i = 0; all.size() < count; i++) {
            boolean added = false;
            for (java.util.List<Exercise> bucket : buckets) {
                if (i < bucket.size()) {
                    all.add(bucket.get(i));
                    added = true;
                    if (all.size() >= count) {
                        break;
                    }
                }
            }
            if (!added) {
                break;
            }
        }
        for (int i = 0; i < all.size(); i++) all.get(i).setOrderIndex(i);
        return all;
    }
